cmake_minimum_required(VERSION 3.20)
project(HearthwakeCore LANGUAGES CXX)
add_executable(core-demo Core/demo.cpp)
target_compile_features(core-demo PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(core-demo PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(core-demo PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME core-behavior COMMAND core-demo --self-test)
add_test(NAME real-demo-entrypoint COMMAND core-demo)
set_tests_properties(real-demo-entrypoint PROPERTIES PASS_REGULAR_EXPRESSION "wood=4 total=4 retry=deduplicated")
add_test(NAME invalid-command COMMAND core-demo --unknown)
set_tests_properties(invalid-command PROPERTIES WILL_FAIL TRUE)
